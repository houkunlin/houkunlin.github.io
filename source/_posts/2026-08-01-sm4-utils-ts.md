有时候需要对在项目中临时使用 SM4 加解密计算一些东西，但是又不想启动程序运行工具，或者不想打开在线工具网页，或者只能在命令行终端等场景使用。如果用Java的话，有可能还需要提前编译好。下面提供一个使用 `Bun` 运行的脚本。

工具使用说明：

```
SM4 国密算法工具 — 密钥生成 / 加解密

用法: bun run scripts/sm4-utils.ts [选项]

选项:
  -m, --mode <ECB|CBC>           加密模式（默认 ECB）
  -k, --key <data>               SM4 密钥（编码格式由 --key-encoding 指定，可选，未指定则自动生成；须 16 字节）
  -i, --iv <data>                初始化向量 IV（CBC 模式使用，编码格式与密钥一致，未指定则自动生成；须 16 字节）
  -e, --encrypt <text>           待加密的明文字符串
  -d, --decrypt <data>           待解密的密文（编码格式由 --cipher-encoding 指定，兼容 SM4ENC(...) 格式）
  -ke, --key-encoding <fmt>      密钥 / IV 编码格式：HEX | BASE64 | BASE64URL | BASE32 | UTF8 | BINARY（默认 HEX；UTF8 仅适用于密钥 / IV）
  -ce, --cipher-encoding <fmt>   密文输入 / 输出编码格式：HEX | BASE64 | BASE64URL | BASE32 | BINARY（默认 HEX，不支持 UTF8）
  -h, --help                     显示此帮助信息

使用示例:
  bun run scripts/sm4-utils.ts -k 0123456789ABCDEFFEDCBA9876543210 -e HelloWorld
  bun run scripts/sm4-utils.ts -k 0123456789ABCDEFFEDCBA9876543210 -d 3F4A...
  bun run scripts/sm4-utils.ts -ke BASE64 -k ASNFZ4mrze8= -ce BASE64 -e HelloWorld
  bun run scripts/sm4-utils.ts -ke UTF8 -k 1234567890abcdef -e HelloWorld
  bun run scripts/sm4-utils.ts -e HelloWorld
  bun run scripts/sm4-utils.ts -m CBC -k 0123456789ABCDEFFEDCBA9876543210 -i 0123456789ABCDEFFEDCBA9876543210 -e HelloWorld
  bun run scripts/sm4-utils.ts -m CBC -k 0123456789ABCDEFFEDCBA9876543210 -d 3F4A...
```

// ======================== 固定常量 CK（32 个 32 位字） ========================

function generateCK(): number[] {
  const ck = new Array<number>(32)
  for (let i = 0; i < 32; i++) {
    const b0 = ((4 * i) * 7) & 0xff
    const b1 = ((4 * i + 1) * 7) & 0xff
    const b2 = ((4 * i + 2) * 7) & 0xff
    const b3 = ((4 * i + 3) * 7) & 0xff
    ck[i] = (b0 << 24) | (b1 << 16) | (b2 << 8) | b3
  }
  return ck
}

const CK = generateCK()

// ======================== PKCS7 填充 ========================

/**
 * PKCS7 填充，每 16 字节一组，不足时补足，满 16 时额外补 16 字节 0x10
 */
function pkcs7Pad(data: Uint8Array): Uint8Array {
  const padLen = 16 - (data.length % 16)
  const padded = new Uint8Array(data.length + padLen)
  padded.set(data)
  padded.fill(padLen, data.length)
  return padded
}

/**
 * 去除 PKCS7 填充
 */
function pkcs7Unpad(data: Uint8Array): Uint8Array {
  const padLen = data[data.length - 1]
  if (padLen < 1 || padLen > 16) {
    throw new Error('无效的 PKCS7 填充')
  }
  return data.slice(0, data.length - padLen)
}

// ======================== 核心算法 ========================

/**
 * 32 位整数的循环左移
 *
 * @param a 待移位整数
 * @param n 左移位数（0-31）
 * @returns 循环左移结果
 */
function rotl(a: number, n: number): number {
  return (a << n) | (a >>> (32 - n))
}

/**
 * S 盒字节替换：对 32 位字的每个字节查 S 盒
 *
 * @param x 32 位输入
 * @returns S 盒替换后的 32 位结果
 */
function sBoxSub(x: number): number {
  return (S_BOX[(x >>> 24) & 0xff] << 24)
    | (S_BOX[(x >>> 16) & 0xff] << 16)
    | (S_BOX[(x >>> 8) & 0xff] << 8)
    | S_BOX[x & 0xff]
}

/**
 * 非线性变换 τ：4 个 S 盒并行替换
 */
function tau(a: number): number {
  return sBoxSub(a)
}

/**
 * 线性变换 L（加密/解密中使用）
 * <p>L(B) = B ⊕ (B <<< 2) ⊕ (B <<< 10) ⊕ (B <<< 18) ⊕ (B <<< 24)</p>
 */
function lTransform(b: number): number {
  return b ^ rotl(b, 2) ^ rotl(b, 10) ^ rotl(b, 18) ^ rotl(b, 24)
}

/**
 * 线性变换 L'（密钥扩展中使用）
 * <p>L'(B) = B ⊕ (B <<< 13) ⊕ (B <<< 23)</p>
 */
function lPrimeTransform(b: number): number {
  return b ^ rotl(b, 13) ^ rotl(b, 23)
}

/**
 * 合成变换 T = L(τ(a))，用于加解密轮函数
 */
function tTransform(a: number): number {
  return lTransform(tau(a))
}

/**
 * 合成变换 T' = L'(τ(a))，用于密钥扩展
 */
function tPrimeTransform(a: number): number {
  return lPrimeTransform(tau(a))
}

/**
 * 轮函数 F
 * <p>X0 ⊕ T(X1 ⊕ X2 ⊕ X3 ⊕ rk)</p>
 */
function roundFunction(x0: number, x1: number, x2: number, x3: number, rk: number): number {
  return x0 ^ tTransform(x1 ^ x2 ^ x3 ^ rk)
}

/**
 * 密钥扩展：从 128 位密钥生成 32 个 32 位轮密钥
 *
 * @param keyBytes 16 字节密钥
 * @returns 32 个轮密钥
 */
function keyExpansion(keyBytes: Uint8Array): number[] {
  const mk = bytesToWords(keyBytes)
  const k0 = mk[0] ^ FK0
  const k1 = mk[1] ^ FK1
  const k2 = mk[2] ^ FK2
  const k3 = mk[3] ^ FK3

  const rk = new Array<number>(32)
  const k = new Array<number>(36)
  k[0] = k0
  k[1] = k1
  k[2] = k2
  k[3] = k3

  for (let i = 0; i < 32; i++) {
    k[i + 4] = k[i] ^ tPrimeTransform(k[i + 1] ^ k[i + 2] ^ k[i + 3] ^ CK[i])
    rk[i] = k[i + 4]
  }
  return rk
}

/**
 * ECB 模式加密一个 16 字节数据块
 *
 * @param plaintext 16 字节明文块
 * @param rk        32 个轮密钥
 * @returns 16 字节密文块
 */
function encryptBlock(plaintext: Uint8Array, rk: number[]): Uint8Array {
  const x = bytesToWords(plaintext)
  for (let i = 0; i < 32; i++) {
    const x4 = roundFunction(x[0], x[1], x[2], x[3], rk[i])
    x[0] = x[1]
    x[1] = x[2]
    x[2] = x[3]
    x[3] = x4
  }
  // 反序输出 X35, X34, X33, X32
  return wordsToBytes([x[3], x[2], x[1], x[0]])
}

/**
 * ECB 模式解密一个 16 字节数据块（使用反向轮密钥）
 */
function decryptBlock(ciphertext: Uint8Array, rk: number[]): Uint8Array {
  const x = bytesToWords(ciphertext)
  // 解密时轮密钥逆序使用
  for (let i = 0; i < 32; i++) {
    const x4 = roundFunction(x[0], x[1], x[2], x[3], rk[31 - i])
    x[0] = x[1]
    x[1] = x[2]
    x[2] = x[3]
    x[3] = x4
  }
  return wordsToBytes([x[3], x[2], x[1], x[0]])
}

/**
 * 两个 16 字节块按位异或（用于 CBC 模式）
 */
function xor16(a: Uint8Array, b: Uint8Array): Uint8Array {
  const out = new Uint8Array(16)
  for (let i = 0; i < 16; i++) {
    out[i] = a[i] ^ b[i]
  }
  return out
}

/**
 * SM4 ECB 模式加密
 *
 * @param plaintext 明文字节数组
 * @param key       16 字节密钥
 * @returns 密文字节数组
 */
export function encryptECB(plaintext: Uint8Array, key: Uint8Array): Uint8Array
/**
 * SM4 ECB 模式加密（密钥为 HEX 格式）
 *
 * @param plaintext 明文字节数组
 * @param keyHex    HEX 格式密钥（32 字符，16 字节）
 * @returns 密文 HEX 字符串
 */
export function encryptECB(plaintext: Uint8Array, keyHex: string): string
export function encryptECB(plaintext: Uint8Array, key: Uint8Array | string): Uint8Array | string {
  const keyBytes = typeof key === 'string' ? hexToBytes(key) : key
  const rk = keyExpansion(keyBytes)
  const padded = pkcs7Pad(plaintext)
  const result = new Uint8Array(padded.length)
  for (let i = 0; i < padded.length; i += 16) {
    const block = padded.slice(i, i + 16)
    const enc = encryptBlock(block, rk)
    result.set(enc, i)
  }
  return typeof key === 'string' ? bytesToHex(result) : result
}

/**
 * SM4 CBC 模式加密（密钥 / IV 为 HEX 格式）
 *
 * @param plaintext 明文字节数组
 * @param keyHex    HEX 格式密钥
 * @param ivHex     HEX 格式初始化向量（32 字符，16 字节）
 * @returns 密文 HEX 字符串

export function encryptCBC(plaintext: Uint8Array, keyHex: string, ivHex: string): string
export function encryptCBC(plaintext: Uint8Array, key: Uint8Array | string, iv: Uint8Array | string): Uint8Array | string {
  const keyBytes = typeof key === 'string' ? hexToBytes(key) : key
  const ivBytes = typeof iv === 'string' ? hexToBytes(iv) : iv
  const rk = keyExpansion(keyBytes)
  const padded = pkcs7Pad(plaintext)
  const result = new Uint8Array(padded.length)

    const enc = encryptBlock(xored, rk)
    result.set(enc, i)
    prev = enc
  }
  return typeof key === 'string' ? bytesToHex(result) : result
}

// ======================== HEX / 字节数组工具 ========================

/**
 * HEX 字符串转字节数组
 */
export function hexToBytes(hex: string): Uint8Array {
  const len = hex.length
  const data = new Uint8Array(len / 2)
  for (let i = 0; i < len; i += 2) {
    data[i / 2] = (parseInt(hex.charAt(i), 16) << 4) | parseInt(hex.charAt(i + 1), 16)
  }
  return data
}

/**
 * 字节数组转大写 HEX 字符串
 */
export function bytesToHex(bytes: Uint8Array): string {
  let sb = ''
  for (const b of bytes) {
    sb += (b & 0xff).toString(16).padStart(2, '0')
  }
  return sb.toUpperCase()
}

// ======================== BASE64 / BASE64URL / BASE32 / UTF8 / BINARY 编码工具 ========================

/**
 * Base32 字母表（RFC 4648，去除 I、L、O、0、1 等易混淆字符）
 */
const BASE32_ALPHABET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567'

/**
 * 字节数组转 BASE64 字符串（RFC 4648，带 `=` 填充）
 */
export function bytesToBase64(bytes: Uint8Array): string {
  return Buffer.from(bytes).toString('base64')
}

/**
 * BASE64 字符串转字节数组（RFC 4648）
 */
export function base64ToBytes(data: string): Uint8Array {
  return new Uint8Array(Buffer.from(data, 'base64'))
}

/**
 * 字节数组转 BASE64URL 字符串（RFC 4648 §5，`-_` 替换 `+/`，带 `=` 填充）
 */
export function bytesToBase64Url(bytes: Uint8Array): string {
  let sb = Buffer.from(bytes).toString('base64url')
  while (sb.length % 4 !== 0) {
    sb += '='
  }
  return sb
}

/**
 * BASE64URL 字符串转字节数组（RFC 4648 §5，兼容 `=` 填充）
 */
export function base64UrlToBytes(data: string): Uint8Array {
  return new Uint8Array(Buffer.from(data.replace(/=+$/, ''), 'base64url'))
}

/**
 * 字节数组转 BASE32 字符串（RFC 4648，带 `=` 填充）
 */
export function bytesToBase32(bytes: Uint8Array): string {
  let buffer = 0
  let bitsLeft = 0
  let sb = ''
  for (let i = 0; i < bytes.length; i++) {
    buffer = (buffer << 8) | bytes[i]
    bitsLeft += 8
    while (bitsLeft >= 5) {
      bitsLeft -= 5
      sb += BASE32_ALPHABET[(buffer >>> bitsLeft) & 0x1f]
    }
  }
  if (bitsLeft > 0) {
    sb += BASE32_ALPHABET[(buffer << (5 - bitsLeft)) & 0x1f]
  }
  while (sb.length % 8 !== 0) {
    sb += '='
  }
  return sb
}

/**
 * BASE32 字符串转字节数组（RFC 4648，兼容 `=` 填充）
 */
export function base32ToBytes(data: string): Uint8Array {
  const cleaned = data.replace(/=+$/, '').toUpperCase()
  const out: number[] = []
  let buffer = 0
  let bitsLeft = 0
  for (let i = 0; i < cleaned.length; i++) {
    const idx = BASE32_ALPHABET.indexOf(cleaned[i])
    if (idx < 0) {
      throw new Error(`非法 BASE32 字符: ${cleaned[i]}`)
    }
    buffer = (buffer << 5) | idx
    bitsLeft += 5
    if (bitsLeft >= 8) {
      bitsLeft -= 8
      out.push((buffer >>> bitsLeft) & 0xff)
    }
  }
  return new Uint8Array(out)
}

/**
 * 字节数组转 BINARY（二进制 0/1 字符串，每字节 8 位）
 */
export function bytesToBinary(bytes: Uint8Array): string {
  let sb = ''
  for (const b of bytes) {
    sb += (b & 0xff).toString(2).padStart(8, '0')
  }
  return sb
}

/**
 * BINARY（二进制 0/1 字符串）转字节数组
 */
export function binaryToBytes(data: string): Uint8Array {
  if (data.length % 8 !== 0) {
    throw new Error('二进制字符串长度必须是 8 的倍数')
  }
  const out = new Uint8Array(data.length / 8)
  for (let i = 0; i < out.length; i++) {
    let val = 0
    for (let j = 0; j < 8; j++) {
      const c = data[i * 8 + j]
      if (c !== '0' && c !== '1') {
        throw new Error(`非法二进制字符: ${c}`)
      }
      val = (val << 1) | (c === '1' ? 1 : 0)
    }
    out[i] = val
  }
  return out
}

/**
 * 密钥 / IV / 密文支持的编码格式
 */
export type Encoding = 'HEX' | 'BASE64' | 'BASE64URL' | 'BASE32' | 'UTF8' | 'BINARY'

/**
 * 按指定编码格式将字节数组编码为字符串
 */
export function encodeBytes(bytes: Uint8Array, encoding: Encoding): string {
  switch (encoding) {
    case 'HEX':
      return bytesToHex(bytes)
    case 'BASE64':
      return bytesToBase64(bytes)
    case 'BASE64URL':
      return bytesToBase64Url(bytes)
    case 'BASE32':
      return bytesToBase32(bytes)
    case 'UTF8':
      return new TextDecoder().decode(bytes)
    case 'BINARY':
      return bytesToBinary(bytes)
  }
}

/**
 * 按指定编码格式将字符串解码为字节数组
 */
export function decodeBytes(data: string, encoding: Encoding): Uint8Array {
  switch (encoding) {
    case 'HEX':
      return hexToBytes(data)
    case 'BASE64':
      return base64ToBytes(data)
    case 'BASE64URL':
      return base64UrlToBytes(data)
    case 'BASE32':
      return base32ToBytes(data)
    case 'UTF8':
      return new TextEncoder().encode(data)
    case 'BINARY':
      return binaryToBytes(data)
  }
}

/**
 * 解析编码格式（不区分大小写），失败时打印错误并退出
 */
function parseEncoding(value: string): Encoding {
  const upper = value.toUpperCase()
  switch (upper) {
    case 'HEX':
    case 'BASE64':
    case 'BASE64URL':
    case 'BASE32':
    case 'UTF8':
    case 'BINARY':
      return upper as Encoding
    default:
      console.error(`错误：不支持的编码格式: ${value}（仅支持 HEX / BASE64 / BASE64URL / BASE32 / UTF8 / BINARY）`)
      process.exit(1)
  }
}

/**
 * 4 字节 → 1 个 int（大端序）
 */
function bytesToWord(b: Uint8Array, off: number): number {
  return (b[off] & 0xff) << 24
    | (b[off + 1] & 0xff) << 16
    | (b[off + 2] & 0xff) << 8
    | (b[off + 3] & 0xff)
}

/**
 * 字节数组 → int 数组（每 4 字节一组）
 */
function bytesToWords(b: Uint8Array): number[] {
  const words = new Array<number>(b.length / 4)
  for (let i = 0; i < words.length; i++) {
    words[i] = bytesToWord(b, i * 4)
  }
  return words
}

/**
 * int 数组（大端序）→ 字节数组
 */
function wordsToBytes(words: number[]): Uint8Array {
  const b = new Uint8Array(words.length * 4)
  for (let i = 0; i < words.length; i++) {
    b[i * 4] = (words[i] >>> 24) & 0xff
    b[i * 4 + 1] = (words[i] >>> 16) & 0xff
    b[i * 4 + 2] = (words[i] >>> 8) & 0xff
    b[i * 4 + 3] = words[i] & 0xff
  }
  return b
}

function main() {
  const args = process.argv.slice(2)
  let key: string | null = null
  let encryptText: string | null = null
  let decryptText: string | null = null
  let iv: string | null = null
  let cbcMode = false
  let showHelp = false
  let keyEncoding: Encoding = 'HEX'
  let cipherEncoding: Encoding = 'HEX'

  for (let i = 0; i < args.length; i++) {
    switch (args[i]) {
      case '-h':
      case '--help':
        showHelp = true
        break
      case '-m':
      case '--mode':
        if (i + 1 >= args.length) {
          console.error('错误：-m 缺少模式参数')
          printUsage()
          return
        }
        const mode = args[++i]
        if (mode.toLowerCase() === 'cbc') {
          cbcMode = true
        } else if (mode.toLowerCase() !== 'ecb') {
          console.error(`错误：不支持的加密模式: ${mode}（仅支持 ECB / CBC）`)
          printUsage()
          return
        }
        break
      case '-i':
      case '--iv':
        if (i + 1 >= args.length) {
          console.error('错误：-i 缺少 IV 参数')
          printUsage()
          return
        }
        iv = args[++i]
        break
      case '-k':
      case '--key':
        if (i + 1 >= args.length) {
          console.error('错误：-k 缺少密钥参数')
          printUsage()
          return
        }
        key = args[++i]
        break
      case '-e':
      case '--encrypt':
        if (i + 1 >= args.length) {
          console.error('错误：-e 缺少明文参数')
          printUsage()
          return
        }
        encryptText = args[++i]
        break
      case '-d':
      case '--decrypt':
        if (i + 1 >= args.length) {
          console.error('错误：-d 缺少密文参数')
          printUsage()
          return
        }
        decryptText = args[++i]
        break
      case '-ke':
      case '--key-encoding':
        if (i + 1 >= args.length) {
          console.error('错误：-ke 缺少编码格式参数')
          printUsage()
          return
        }
        keyEncoding = parseEncoding(args[++i])
        break
      case '-ce':
      case '--cipher-encoding':
        if (i + 1 >= args.length) {
          console.error('错误：-ce 缺少编码格式参数')
          printUsage()
          return
        }
        cipherEncoding = parseEncoding(args[++i])
        if (cipherEncoding === 'UTF8') {
          console.error('错误：UTF8 明文编码仅适用于密钥 / IV（--key-encoding），密文编码请使用 HEX / BASE64 / BASE64URL / BASE32 / BINARY')
          printUsage()
          return
        }
        break
      default:
        // 忽略未知参数
        break
    }
  }

  if (showHelp || (encryptText === null && decryptText === null)) {
    printUsage()
    return
  }

  let keyBytes: Uint8Array
  let ivBytes: Uint8Array | null = null
  try {
    if (key === null) {
      keyBytes = generateKeyBytes()
      console.log(`密钥（自动生成）：${encodeBytes(keyBytes, keyEncoding)}`)
    } else {
      keyBytes = decodeBytes(key, keyEncoding)
      console.log(`密钥：${encodeBytes(keyBytes, keyEncoding)}`)
    }
    if (keyBytes.length !== 16) {
      console.error(`错误：SM4 密钥必须为 16 字节，当前为 ${keyBytes.length} 字节`)
      return
    }
    if (cbcMode) {
      if (iv === null) {
        ivBytes = generateIvBytes()
        console.log(`IV（自动生成）：${encodeBytes(ivBytes, keyEncoding)}`)
      } else {
        ivBytes = decodeBytes(iv, keyEncoding)
        console.log(`IV：${encodeBytes(ivBytes, keyEncoding)}`)
      }
      if (ivBytes.length !== 16) {
        console.error(`错误：IV 必须为 16 字节，当前为 ${ivBytes.length} 字节`)
        return
      }
    }
  } catch (e) {
    console.error(`密钥 / IV 解码失败：${(e as Error).message}`)
    return
  }

  const modeLabel = cbcMode ? 'CBC' : 'ECB'

  try {
    if (encryptText !== null) {
      const encrypted = cbcMode
        ? encryptCBC(new TextEncoder().encode(encryptText), keyBytes, ivBytes!)
    : encryptECB(new TextEncoder().encode(encryptText), keyBytes)
      console.log(`加密模式：${modeLabel}`)
      console.log(`加密明文：${encryptText}`)
      console.log(`加密密文：${encodeBytes(encrypted, cipherEncoding)}`)
    }
  } catch (e) {
    console.error(`SM4 加密失败：${(e as Error).message}`)
  }

  try {
    if (decryptText !== null) {
      let plain = decryptText
      if (plain.startsWith('SM4ENC(') && plain.endsWith(')')) {
        plain = plain.substring(7, plain.length - 1)
      }
      const cipherBytes = decodeBytes(plain, cipherEncoding)
      const plainBytes = cbcMode
        ? decryptCBC(cipherBytes, keyBytes, ivBytes!)
    : decryptECB(cipherBytes, keyBytes)
      console.log(`解密模式：${modeLabel}`)
      console.log(`解密密文：${decryptText}`)
      console.log(`解密明文：${new TextDecoder().decode(plainBytes)}`)
    }
  } catch (e) {
    console.error(`SM4 解密失败：${(e as Error).message}`)
  }
}

  main()
}
